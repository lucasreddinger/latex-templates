\section{Model}
\subsection{Setup}

% \framesubtitle adds a line under the frame title.
\begin{frame}{Quasi-hyperbolic discounting}
  \framesubtitle{Present bias in one parameter}
  A decision maker values a stream of consumption as
  \[
    U_t = u(x_t) + \beta \sum_{\tau=1}^{T} \delta^\tau u(x_{t+\tau})
  \]
  \begin{itemize}
    \item $\beta = 1$: exponential discounting
    \item $\beta < 1$: present bias
  \end{itemize}
  \begin{exampleblock}{Example}
    With $\beta = 0.7$ and $\delta = 1$, \$110 tomorrow is worth \$77 today,
    so \$100 today wins. A week earlier, the same person prefers the \$110.
  \end{exampleblock}
\end{frame}

\subsection{Predictions}

% Beamer's built-in theorem environments (theorem, lemma, corollary,
% definition, example) and proof all work without extra setup.
\begin{frame}{The value of commitment}
  \begin{theorem}
    A present-biased person who knows that $\beta < 1$ strictly prefers to
    remove a tempting early option in advance.
  \end{theorem}
  \begin{proof}[Proof sketch]
    Ahead of time, the person ranks the options using $\delta$ alone; at the
    moment of choice, the extra factor $\beta$ reverses that ranking.
    Removing the early option prevents the reversal.
  \end{proof}
\end{frame}

\begin{frame}{Hypotheses}
  \begin{block}{Hypothesis 1}
    The treatment lowers present bias:
    $\beta_{\text{treat}} < \beta_{\text{control}}$
  \end{block}
  \pause
  \begin{block}{Hypothesis 2}
    The treatment raises take-up of the commitment device
  \end{block}
\end{frame}

\section{Design}

% A description list. The optional [Commitment] sets the label width to the
% longest label, so the entries line up.
\begin{frame}{Experimental design}
  \begin{description}[Commitment]
    \item[Control] Rewards shown as calendar dates
    \item[Treatment] Rewards shown as delays from today
    \item[Commitment] An option to lock in the later reward in advance
  \end{description}
  \medskip
  897 participants, randomly assigned to the two
  conditions.\footnote{Pre-registered at the AEA RCT Registry
  (AEARCTR-0000000).}
\end{frame}

\begin{frame}{Estimation}
  \begin{enumerate}
    \item Estimate $\beta$ and $\delta$ for each participant by maximum
    likelihood
    \item Compare $\beta$ across conditions, with standard errors that
    account for the two-stage estimator
  \end{enumerate}
\end{frame}
