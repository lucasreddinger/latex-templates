\section{Introduction}

\begin{frame}{Motivation}
  % [<+->] reveals one item per click.
  \begin{itemize}[<+->]
    \item People plan to act patiently, then take the immediate reward when
    the moment arrives \parencite{Doe2020,RoeSmith2019}.
    \item \textcite{LeeKimPark2021} model this as \alert{present bias}: an
    extra discount on everything that is not ``now.''
    \item As \citeauthor{Doe2020} (\citeyearlinked{Doe2020}) shows, people who
    know they are present-biased seek commitment.
  \end{itemize}
\end{frame}

\begin{frame}{This paper}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item Laboratory experiment: a sequence of intertemporal allocation
        decisions under one of two conditions
        \item Estimate a quasi-hyperbolic discounting model
        \item \alert{Finding:} the treatment lowers the present-bias
        parameter~$\beta$ and raises demand for commitment
      \end{itemize}
    \end{column}
    \begin{column}{0.4\textwidth}
      \centering
      \begin{tikzpicture}
        \draw[thick] (0,0) rectangle (5,3);
        \node at (2.5,1.5) {design schematic};
      \end{tikzpicture}
    \end{column}
  \end{columns}
\end{frame}

\section{Model}

\begin{frame}{Quasi-hyperbolic discounting}
  A decision maker values a stream of consumption as
  \begin{align}
      U_t &= u(x_t) + \beta \sum_{\tau=1} \delta^\tau\, u(x_{t+\tau}).
      \label{eq:objective}
  \end{align}
  \begin{itemize}
    \item $\beta = 1$: exponential discounting; $\beta < 1$: present bias
    \item Take-up of the commitment device, $\mybb{1}(\text{commit})$, rises as
    $\beta$ in \cref{eq:objective} falls
  \end{itemize}
\end{frame}

\begin{frame}{Hypotheses}
  \begin{hypothesis}\label{hyp:main}
  The treatment lowers the present-bias parameter relative to the control:
  $\beta_{\text{treat}} < \beta_{\text{control}}$.
  \end{hypothesis}
  \pause
  \begin{hypothesis}\label{hyp:secondary}
  Participants in the treatment are more likely to take up the commitment
  device.
  \end{hypothesis}
\end{frame}

\begin{frame}{Estimation}
  % Beamer's own label syntax: the "1" is replaced by the item number. Beamer
  % does not widen the indent for long labels, so set it to the label's width.
  \settowidth{\leftmargini}{\bfseries Step 1:\hspace{\labelsep}}
  \begin{enumerate}[\bfseries Step 1:]
    \item Estimate $\beta$ and $\delta$ for each participant by maximum
    likelihood from the allocation choices.
    \item Compare the estimated $\beta$ across conditions, reporting
    \textcitedash{RoeSmith2019} standard errors to account for the two-stage
    estimator.
  \end{enumerate}
  \medskip
  This tests \cref{hyp:main,hyp:secondary}.
\end{frame}

\section{Conclusion}

\begin{frame}{Takeaways}
  \begin{itemize}
    \item A simple change in how future rewards are presented shifts the
    estimated present-bias parameter
    \item It also raises demand for commitment
    \item Caveat: a single laboratory sample; field replication is the natural
    next step
  \end{itemize}
\end{frame}
