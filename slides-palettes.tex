% \usepalette{paper}{ink}{accent}{highlight}{closing}: the five palette colors
% as hex codes (the values in the template's palette blocks), for the frames
% that follow.
\newcommand{\usepalette}[5]{%
  \definecolor{paper}{HTML}{#1}\definecolor{ink}{HTML}{#2}%
  \definecolor{accent}{HTML}{#3}\definecolor{highlight}{HTML}{#4}%
  \definecolor{closing}{HTML}{#5}%
  \def\palettecodes{Background \##1\quad Text \##2\quad Rules and bullets
    \##3\\Highlights \##4\quad Closing slide \##5}}
\usepalette{FAFAFA}{23373B}{23373B}{D55E00}{333333}

% Colors and frame titles as in slides-template.tex.
\setbeamercolor{normal text}{fg=ink,bg=paper}
\setbeamercolor{structure}{fg=accent}
\setbeamercolor{frametitle}{fg=ink,bg=}
\setbeamercolor{block title}{fg=ink,bg=accent!15}
\setbeamercolor{block body}{bg=accent!5}
\setbeamercolor{alerted text}{fg=highlight}
\setbeamerfont{alerted text}{series=\bfseries}

\newcommand{\titlerule}{%
  \par\kern-\prevdepth\kern6pt\nointerlineskip
  \noindent{\color{accent}\rule{\textwidth}{0.4pt}}\par}
\setbeamertemplate{frametitle}{%
  \begin{beamercolorbox}[wd=\paperwidth,leftskip=1cm,rightskip=1cm plus 1fil]{frametitle}%
    \rule{0pt}{4ex}\insertframetitle\par
    {\usebeamerfont{framesubtitle}\insertframesubtitle\par}%
    \titlerule
  \end{beamercolorbox}%
}

% No page numbers: the bottom-right corner holds the closing-slide tile.
\setbeamertemplate{page number in head/foot}{}

% \showpalette{name}{paper}{ink}{accent}{highlight}{closing}: one sample
% frame in that palette. Bottom left: the color codes. Bottom right: a small
% tile in the closing slide's colors. Both sit 1cm in from the sides.
\newcommand{\showpalette}[6]{%
  \usepalette{#2}{#3}{#4}{#5}{#6}%
  \frame{\frametitle{#1}%
    \begin{columns}[T,onlytextwidth]
      \begin{column}{0.48\textwidth}
        \begin{itemize}
          \item People plan to act patiently
          \item \alert{Present bias:} an extra discount on ``now''
          \item Demand for commitment rises
        \end{itemize}
        \medskip
        \begin{block}{Hypothesis 1}
          The treatment lowers present bias:
          $\beta_{\text{treat}} < \beta_{\text{control}}$
        \end{block}
      \end{column}
      \begin{column}{0.48\textwidth}
        \centering
        \begin{tabular}{lcc}
          \toprule
                           & Model 1 & Model 2 \\
          \midrule
          Treatment effect & 0.281   & 0.895   \\
                           & (0.044) & (0.052) \\
          \bottomrule
        \end{tabular}

        \bigskip
        \beamergotobutton{Robustness}
      \end{column}
    \end{columns}
    \begin{tikzpicture}[remember picture,overlay]
      \node[anchor=south west,align=left,inner sep=0pt,font=\footnotesize]
        at ([xshift=1cm,yshift=0.8cm]current page.south west) {\palettecodes};
      \node[anchor=south east,fill=closing,text=paper,font=\small\bfseries,
            minimum width=3.6cm,minimum height=2.025cm]
        at ([xshift=-1cm,yshift=0.8cm]current page.south east) {Thank you};
    \end{tikzpicture}}%
}
